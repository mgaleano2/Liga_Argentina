Import:
    from ScraperFC.sofascore import Sofascore

--------------------------------------
1) FUNCIONES POR PARTIDO (match-level)
--------------------------------------

get_match_dict(match_id: str | int) -> dict
    Devuelve el dict crudo de un partido: equipos, resultado, fecha
    (startTimestamp), ronda (roundInfo.round), torneo, etc.

get_match_dicts(year: str, league: str) -> list[dict]
    Devuelve TODOS los partidos de una temporada de una liga soportada.
    Pagina internamente el endpoint events/last/{i} del season.
    Útil para armar ventanas de "últimas N fechas" por equipo.

get_match_id_from_url(match_url: str) -> int
    Extrae el match id desde una URL de Sofascore.

get_match_url_from_id(match_id: str | int) -> str
    Genera la URL de Sofascore para un match id.

get_match_player_ids(match_id: str | int) -> dict
    Devuelve los IDs de todos los jugadores del partido
    (estructura: home/away).

get_team_names(match_id: str | int) -> tuple[str, str]
    Devuelve (nombre_local, nombre_visitante).

scrape_player_match_stats(match_id: str | int) -> pandas.DataFrame
    Stats de cada jugador del partido. Columnas principales:
      name, teamName, position (G/D/M/F), minutesPlayed, rating,
      goals, goalAssist, expectedGoals, expectedAssists, totalShots,
      onTargetScoringAttempt, totalPass, accuratePass,
      accurateOwnHalfPasses / totalOwnHalfPasses,
      accurateOppositionHalfPasses / totalOppositionHalfPasses,
      totalLongBalls / accurateLongBalls, keyPass, bigChanceCreated,
      aerialWon / aerialLost, duelWon / duelLost, totalTackle,
      interceptionWon, totalClearance, ballRecovery, saves, ...
    OJO: el DF viene SIN columna de match_id (hay que agregarla a mano)
    y la columna "position" puede aparecer duplicada (position.1) al
    aplanarse el dict del jugador — usar la primera.

scrape_team_match_stats(match_id: str | int) -> pandas.DataFrame
    Stats generales del partido por equipo.

scrape_match_shots(match_id: str | int) -> pandas.DataFrame
    Tiros del partido (coordenadas x/y, xG por tiro).

scrape_match_momentum(match_id: str | int) -> pandas.DataFrame
    Valores de momentum del partido.

scrape_heatmaps(match_id: str | int) -> dict
    Coordenadas x-y de todos los jugadores para dibujar heatmaps.

scrape_player_average_positions(match_id: str | int) -> pandas.DataFrame
    Posición promedio de cada jugador en el partido.

------------------------------------------------
2) FUNCIONES POR LIGA / TEMPORADA (league-level)
------------------------------------------------

get_valid_seasons(league: str) -> dict
    Devuelve {year: season_id} de las temporadas disponibles de una liga.

get_positions(selected_positions: list[str]) -> str
    Traduce ["Goalkeepers","Defenders","Midfielders","Forwards"] a los
    filtros de la API (G, D, M, F unidos por ~).

scrape_player_league_stats(year: str, league: str,
                           accumulation: str = "total",
                           selected_positions: list[str] = [...]) -> pandas.DataFrame
    Stats de TODA la temporada para los jugadores de la liga, filtradas
    por posición.
    - accumulation: "total" | "per90" | "perMatch"
    - selected_positions: ["Goalkeepers","Defenders","Midfielders","Forwards"]
    Devuelve: nombre, equipo, rating, countRating (cantidad de ratings =
    PJ), appearances, minutesPlayed, goals, assists, expectedGoals,
    expectedAssists, totalShots, keyPasses, etc.
    OJO: es de TEMPORADA COMPLETA. No hay filtro de "últimas N fechas".
    OJO: para separar por posición hay que llamarla UNA VEZ POR POSICIÓN
    (el DF resultante no trae columna de posición).

scrape_team_league_stats(year: str, league: str) -> pandas.DataFrame
    Stats generales de todos los equipos de la liga para la temporada.

scrape_player_details(year: str, league: str) -> list[SofascorePlayer]
    Detalles de todos los jugadores de la liga: nombre, DOB, posición,
    altura, peso, fin de contrato, etc.
    OJO: el equipo es el ACTUAL del jugador, no necesariamente el de la
    temporada consultada.

5) EJEMPLO RÁPIDO — Top 20 por posición (temporada completa)
-----------------------------------------------------------

from ScraperFC.sofascore import Sofascore
import pandas as pd

s = Sofascore()
POS = {"Defenders": "defensores", "Midfielders": "mediocampistas", "Forwards": "delanteros"}

for pos_en, pos_es in POS.items():
    df = s.scrape_player_league_stats("2026", "Argentina Liga Profesional",
                                      accumulation="total", selected_positions=[pos_en])
    df["pj"] = df["countRating"]
    df = df[df["pj"] > 5]
    df = df.sort_values(["rating", "minutesPlayed"], ascending=[False, False]).head(20)
    print(f"\n=== TOP 20 {pos_es.upper()} ===")
    print(df[["player", "team", "pj", "rating", "goals", "assists"]].to_string(index=False))
